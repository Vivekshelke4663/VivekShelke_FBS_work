void main()
{
    char str[100], ch, symbol;
    int i;

    printf("Enter string: ");
    scanf("%s", str);

    printf("Enter character: ");
    scanf(" %c", &ch);

    printf("Enter special symbol: ");
    scanf(" %c", &symbol);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == ch)
            str[i] = symbol;
    }

    printf("New string = %s", str);

}
